        // localStorage 辅助函数
        function safeLocalStorageGet(key, defaultValue = null) {
            try {
                const value = localStorage.getItem(key);
                return value ? JSON.parse(value) : defaultValue;
            } catch (error) {
                return defaultValue;
            }
        }

        function safeLocalStorageSet(key, value) {
            try {
                localStorage.setItem(key, JSON.stringify(value));
                return true;
            } catch (error) {
                return false;
            }
        }
        
        // 加载用户答题记录
        function loadUserProgress() {
            const saved = safeLocalStorageGet('userProgress', {
                mathQuestions: {},
                speechTypes: {},
                masteredQuestions: [],
                masteredSpeechTypes: []
            });
            userProgress = saved;
        }
        
        // 保存用户答题记录
        function saveUserProgress() {
            safeLocalStorageSet('userProgress', userProgress);
        }
        
        // 更新数学进度

            const masteredCount = userProgress.masteredQuestions.length;
            const totalQuestions = 75;

            document.getElementById('math-mastered').textContent = masteredCount;
            document.getElementById('math-total').textContent = totalQuestions;
            document.getElementById('math-percent').textContent = `${Math.round(progress)}%`;
