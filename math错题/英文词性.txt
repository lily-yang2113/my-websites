        // 初始化
        function init() {
            // 检查是否是同一天
            const today = new Date().toDateString();
            const savedDate = localStorage.getItem('englishPracticeDate');
            
            if (savedDate !== today) {
                // 新的一天，重置每日进度
                resetDailyProgress();
                localStorage.setItem('englishPracticeDate', today);
            } else {
                // 同一天，加载已保存的进度
                loadDailyProgress();
            }
            
            // 加载用户答题记录
            loadUserProgress();
            
            // 更新词性掌握进度
            updateMasteryProgress();
            
            // 创建词性卡片
            createSpeechCards();
            
            // 绑定事件
            bindEvents();
        }

        // 创建词性卡片
        function createSpeechCards() {
            partOfSpeechGrid.innerHTML = '';
            
            partOfSpeechData.forEach(speech => {
                const card = document.createElement('div');
                card.className = 'speech-card';
                card.dataset.speechId = speech.id;
                
                // 检查是否已掌握
                const isMastered = userProgress.masteredSpeechTypes.includes(speech.id);
                const masteryText = isMastered ? '✓ 已掌握' : `掌握度: ${speech.mastery}%`;
                
                card.innerHTML = `
                    <div class="speech-icon" style="background: ${speech.color}">${speech.icon}</div>
                    <div class="speech-name">${speech.name}</div>
                    <div class="speech-desc">${speech.description}</div>
                    <div class="speech-mastery">${masteryText}</div>
                `;
                
                card.addEventListener('click', () => startSpeechPractice(speech.id));
                partOfSpeechGrid.appendChild(card);
            });
        }

        // 加载用户答题记录
        function loadUserProgress() {
            const saved = localStorage.getItem('englishUserProgress');
            if (saved) {
                userProgress = JSON.parse(saved);
            } else {
                // 初始化用户进度
                userProgress = {
                    speechTypes: {},
                    masteredSpeechTypes: []
                };
                
                // 初始化每个词性的进度
                partOfSpeechData.forEach(speech => {
                    userProgress.speechTypes[speech.id] = {
                        correct: 0,
                        total: 0,
                        lastScore: 0
                    };
                });
                
                saveUserProgress();
            }
        }

        // 保存用户答题记录
        function saveUserProgress() {
            localStorage.setItem('englishUserProgress', JSON.stringify(userProgress));
        }

        // 重置每日进度
        function resetDailyProgress() {
            gameState = {
                currentSpeechType: '',
                currentQuestionIndex: 0,
                totalQuestions: 5,
                correctCount: 0,
                wrongCount: 0,
                selectedOption: null,
                practiceCompleted: false,
                currentQuestions: [],
                userAnswers: {},
                todayDate: new Date().toDateString()
            };
            
            saveDailyProgress();
        }

        // 加载每日进度
        function loadDailyProgress() {
            const saved = localStorage.getItem('englishDailyPracticeData');
            if (saved) {
                const data = JSON.parse(saved);
                if (data.todayDate === new Date().toDateString()) {
                    gameState = data;
                } else {
                    resetDailyProgress();
                }
            }
        }

        // 保存每日进度
        function saveDailyProgress() {
            localStorage.setItem('englishDailyPracticeData', JSON.stringify(gameState));
        }

        // 更新词性掌握进度
        function updateMasteryProgress() {
            // 计算已掌握的词性数
            const masteredCount = userProgress.masteredSpeechTypes.length;
            const totalSpeechTypes = partOfSpeechData.length;
            
            masteryCompleted.textContent = masteredCount;
            masteryTotal.textContent = totalSpeechTypes;
            
            const progress = (masteredCount / totalSpeechTypes) * 100;
            masteryProgressValue.textContent = `${Math.round(progress)}%`;
            masteryProgressFill.style.width = `${progress}%`;
        }

            // 返回按钮
            backBtn.addEventListener('click', returnToHome);
            
            // 提交答案按钮

            // 上一题/下一题按钮
            prevBtn.addEventListener('click', previousQuestion);
            nextBtn.addEventListener('click', nextQuestion);
            
            // 完成界面按钮
            returnBtn.addEventListener('click', returnToHome);
            reviewBtn.addEventListener('click', reviewWrongQuestions);
        }

        // 开始词性练习
        function startSpeechPractice(speechId) {
            gameState.currentSpeechType = speechId;
            gameState.currentQuestionIndex = 0;
            gameState.selectedOption = null;
            gameState.practiceCompleted = false;
            
            // 获取当前词性的题目
            const speech = partOfSpeechData.find(s => s.id === speechId);
            gameState.currentQuestions = [...speech.questions]; // 创建副本
            
            // 打乱题目顺序（但保持第一次的顺序一致）
            const savedOrder = JSON.parse(localStorage.getItem(`${speechId}_question_order`) || 'null');
            const savedDate = localStorage.getItem(`${speechId}_question_order_date`);
            
            if (savedOrder && savedDate === new Date().toDateString()) {
                // 使用之前保存的顺序
                gameState.currentQuestions = savedOrder.map(id => 
                    gameState.currentQuestions.find(q => q.id === id)
                ).filter(q => q); // 过滤掉可能不存在的题目
            } else {
                // 新的一天，随机排序
                shuffleArray(gameState.currentQuestions);
                localStorage.setItem(`${speechId}_question_order`, JSON.stringify(gameState.currentQuestions.map(q => q.id)));
                localStorage.setItem(`${speechId}_question_order_date`, new Date().toDateString());
            }
            
            gameState.userAnswers = {};
            
            // 显示练习区域
            practiceArea.style.display = 'flex';
            backBtn.style.display = 'flex';
            
            // 加载第一题
            loadQuestion(0);
        }

        // 随机打乱数组
        function shuffleArray(array) {
            for (let i = array.length - 1; i > 0; i--) {
                const j = Math.floor(Math.random() * (i + 1));
                [array[i], array[j]] = [array[j], array[i]];
            }
            return array;
        }

        // 加载题目
        function loadQuestion(index) {
            if (index >= gameState.currentQuestions.length) {
                completePractice();
                return;
            }
            
            const question = gameState.currentQuestions[index];
            const speech = partOfSpeechData.find(s => s.id === gameState.currentSpeechType);
            
            // 构建题目HTML
            let html = `
                <div class="question-card">
                    <div class="question-header">
                        <div class="question-number">第 ${index + 1} 题 / 共 5 题</div>
                        <div class="question-type">${speech.name}</div>
                    </div>
                    <div class="question-text">${question.question}</div>
                    <div class="options-container">
            `;
            
            // 添加选项
            const optionLabels = ['A', 'B', 'C', 'D'];
            question.options.forEach((option, i) => {
                const isSelected = gameState.userAnswers[question.id] === i;
                const optionClass = isSelected ? 'option-btn selected' : 'option-btn';
                
                html += `
                    <div class="${optionClass}" data-index="${i}" onclick="selectOption(${i})">
                        <div class="option-label">${optionLabels[i]}</div>
                        <div class="option-text">${option}</div>
                    </div>
                `;
            });
            
            html += `
                    </div>
                    <div class="explanation-area" id="explanation-${question.id}">
                        <div class="explanation-title">📚 详细解析</div>
                        <div class="explanation-text">${question.explanation}</div>
                        <div class="speech-rule">
                            <div class="speech-rule-title">💡 ${speech.name} 用法规则</div>
                            <div class="speech-rule-content">${question.rule}</div>
                        </div>
                    </div>
                </div>
            `;
            
            questionContainer.innerHTML = html;
            
            // 隐藏解析
            const explanationEl = document.getElementById(`explanation-${question.id}`);
            if (explanationEl) {
                explanationEl.style.display = 'none';
            }
            
            // 更新按钮状态
            updateButtonStates(index);
            
            // 保存当前题目索引
            gameState.currentQuestionIndex = index;
            gameState.selectedOption = gameState.userAnswers[question.id] || null;
        }

        // 选择选项
        function selectOption(index) {
            // 如果已经提交过答案，不允许重新选择
            const currentQuestion = gameState.currentQuestions[gameState.currentQuestionIndex];
            if (currentQuestion && currentQuestion.userAnswered) return;
            
            // 清除之前的选择
            document.querySelectorAll('.option-btn').forEach(btn => {
                btn.classList.remove('selected');
            });
            
            // 标记当前选择
            const selectedBtn = document.querySelector(`[data-index="${index}"]`);
            if (selectedBtn) {
                selectedBtn.classList.add('selected');
            }
            
            gameState.selectedOption = index;
            gameState.userAnswers[currentQuestion.id] = index;
        }

        // 提交答案
        function submitAnswer() {
            const question = gameState.currentQuestions[gameState.currentQuestionIndex];
            
            if (gameState.selectedOption === null) {
                alert('请先选择一个答案');
                return;
            }
            
            // 标记题目已回答
            question.userAnswered = true;
            
            // 显示所有选项的正确/错误状态
            document.querySelectorAll('.option-btn').forEach((btn, i) => {
                if (i === question.correct) {
                    btn.classList.add('correct');
                } else if (i === gameState.selectedOption && i !== question.correct) {
                    btn.classList.add('wrong');
                }
            });
            
            // 显示解析
            const explanationEl = document.getElementById(`explanation-${question.id}`);
            if (explanationEl) {
                explanationEl.style.display = 'block';
                explanationEl.classList.add('show');
            }
            
            // 更新统计
            const isCorrect = gameState.selectedOption === question.correct;
            if (isCorrect) {
                gameState.correctCount++;
            } else {
                gameState.wrongCount++;
            }
            
            // 更新用户答题记录
            updateSpeechHistory(isCorrect);
            
            // 保存进度
            saveDailyProgress();
            saveUserProgress();
            
            // 更新按钮状态
            submitBtn.style.display = 'none';
            nextBtn.style.display = 'flex';
            
            // 如果是最后一题，修改按钮文字
            if (gameState.currentQuestionIndex === gameState.currentQuestions.length - 1) {
                nextBtn.innerHTML = '完成练习 <i class="fas fa-flag-checkered"></i>';
            }
        }

        // 更新词性历史记录
        function updateSpeechHistory(isCorrect) {
            const speechId = gameState.currentSpeechType;
            
            // 更新该词性的答题记录
            if (!userProgress.speechTypes[speechId]) {
                userProgress.speechTypes[speechId] = {
                    correct: 0,
                    total: 0,
                    lastScore: 0
                };
            }
            
            userProgress.speechTypes[speechId].total++;
            if (isCorrect) {
                userProgress.speechTypes[speechId].correct++;
            }
            
            // 计算正确率
            const correctRate = (userProgress.speechTypes[speechId].correct / userProgress.speechTypes[speechId].total) * 100;
            
            // 如果正确率达到80%且总答题数达到5次，标记为已掌握
            if (correctRate >= 80 && userProgress.speechTypes[speechId].total >= 5) {
                if (!userProgress.masteredSpeechTypes.includes(speechId)) {
                    userProgress.masteredSpeechTypes.push(speechId);
                    
                    // 更新词性卡片
                    updateSpeechCard(speechId, true);
                    
                    // 更新词性掌握进度
                    updateMasteryProgress();
                }
            }
        }

        // 更新词性卡片
        function updateSpeechCard(speechId, isMastered) {
            const speechCard = document.querySelector(`.speech-card[data-speech-id="${speechId}"]`);
            if (speechCard) {
                const speech = partOfSpeechData.find(s => s.id === speechId);
                const masteryText = isMastered ? '✓ 已掌握' : `掌握度: ${Math.round((userProgress.speechTypes[speechId]?.correct || 0) / (userProgress.speechTypes[speechId]?.total || 1) * 100)}%`;
                
                const masteryEl = speechCard.querySelector('.speech-mastery');
                if (masteryEl) {
                    masteryEl.textContent = masteryText;
                }
            }
        }

        // 更新按钮状态
        function updateButtonStates(index) {
            // 上一题按钮
            prevBtn.disabled = index === 0;
            
            // 提交/下一题按钮
            const question = gameState.currentQuestions[index];
            const isAnswered = question && question.userAnswered;
            
            if (isAnswered) {
                submitBtn.style.display = 'none';
                nextBtn.style.display = 'flex';
                
                // 显示解析
                if (question) {
                    const explanation = document.getElementById(`explanation-${question.id}`);
                    if (explanation) {
                        explanation.style.display = 'block';
                        explanation.classList.add('show');
                    }
                }
            } else {
                submitBtn.style.display = 'flex';
                nextBtn.style.display = 'none';
            }
            
            // 如果是最后一题且已回答，修改下一题按钮文字
            if (index === gameState.currentQuestions.length - 1 && isAnswered) {
                nextBtn.innerHTML = '完成练习 <i class="fas fa-flag-checkered"></i>';
            }
        }

        // 上一题
        function previousQuestion() {
            if (gameState.currentQuestionIndex > 0) {
                loadQuestion(gameState.currentQuestionIndex - 1);
            }
        }

        // 下一题
        function nextQuestion() {
            if (gameState.currentQuestionIndex < gameState.currentQuestions.length - 1) {
                loadQuestion(gameState.currentQuestionIndex + 1);
            } else {
                completePractice();
            }
        }

        // 完成练习
        function completePractice() {
            gameState.practiceCompleted = true;
            
            // 计算统计
            const total = gameState.totalQuestions;
            const correct = gameState.correctCount;
            const accuracy = total > 0 ? Math.round((correct / total) * 100) : 0;
            
            // 检查是否掌握该词性
            const speechId = gameState.currentSpeechType;
            const speech = partOfSpeechData.find(s => s.id === speechId);
            const isMastered = userProgress.masteredSpeechTypes.includes(speechId);
            
            // 更新完成界面
            finalCorrect.textContent = correct;
            finalTotal.textContent = total;
            finalAccuracy.textContent = `${accuracy}%`;
            finalMastered.textContent = isMastered ? '已掌握' : '继续努力';
            
            // 根据表现设置不同的消息
            if (accuracy >= 90) {
                completionTitle.textContent = '太棒了！词性小专家！';
                completionMessage.textContent = `正确率${accuracy}%，你已经掌握了${speech.name}的用法！`;
            } else if (accuracy >= 70) {
                completionTitle.textContent = '做得非常好！';
                completionMessage.textContent = `正确率${accuracy}%，你对${speech.name}的理解很不错！`;
            } else if (accuracy >= 50) {
                completionTitle.textContent = '继续加油！';
                completionMessage.textContent = `正确率${accuracy}%，再多练习几次就能掌握${speech.name}了！`;
            } else {
                completionTitle.textContent = '坚持就是胜利！';
                completionMessage.textContent = `正确率${accuracy}%，多练习一定会更好！记得查看每道题的详细解析哦！`;
            }
            
            // 显示完成界面
            practiceArea.style.display = 'none';
            completionScreen.style.display = 'flex';
            
            // 保存最终进度
            saveDailyProgress();
            saveUserProgress();
        }

        // 返回首页
        function returnToHome() {
            // 隐藏所有界面
            practiceArea.style.display = 'none';
            completionScreen.style.display = 'none';
            backBtn.style.display = 'none';
            
            // 重置游戏状态（但不重置每日进度）
            gameState.currentSpeechType = '';
            gameState.currentQuestionIndex = 0;
            gameState.selectedOption = null;
            
            // 更新词性卡片状态
            createSpeechCards();
            updateMasteryProgress();
        }

        // 复习错题
        function reviewWrongQuestions() {
            // 收集错题
            const wrongQuestions = [];
            
            gameState.currentQuestions.forEach((question, index) => {
                const userAnswer = gameState.userAnswers[question.id];
                if (userAnswer !== undefined && userAnswer !== question.correct) {
                    wrongQuestions.push({
                        question: question,
                        index: index + 1
                    });
                }
            });
            
            if (wrongQuestions.length === 0) {
                alert('恭喜！没有错题需要复习！');
                returnToHome();
                return;
            }
            
            // 重置游戏状态，只练习错题
            gameState.currentQuestions = wrongQuestions.map(item => item.question);
            gameState.totalQuestions = wrongQuestions.length;
            gameState.currentQuestionIndex = 0;
            gameState.selectedOption = null;
            gameState.practiceCompleted = false;
            
            // 重置用户答案（允许重新回答）
            gameState.userAnswers = {};
            gameState.currentQuestions.forEach(q => {
                q.userAnswered = false;
            });
            
            // 重置正确/错误计数
            gameState.correctCount = 0;
            gameState.wrongCount = 0;
            
            // 显示练习区域
            completionScreen.style.display = 'none';
            practiceArea.style.display = 'flex';
            
            // 加载第一题
            loadQuestion(0);
        }

        // 页面加载时初始化
        document.addEventListener('DOMContentLoaded', init);
        
        // 使函数在全局可访问（用于HTML中的onclick）
        window.selectOption = selectOption;
    </script>
