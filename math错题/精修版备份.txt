        // localStorage 辅助函数
        function safeLocalStorageGet(key, defaultValue = null) {
            try {
                const value = localStorage.getItem(key);
                return value ? JSON.parse(value) : defaultValue;
            } catch (error) {
                return defaultValue;
            }
        }

        function safeLocalStorageSet(key, value) {
            try {
                localStorage.setItem(key, JSON.stringify(value));
                return true;
            } catch (error) {
                return false;
            }
        }

        // 为触摸设备添加事件监听
        function setupTouchEvents() {
            // 获取所有可点击元素
            const clickableElements = document.querySelectorAll('.subject-btn, .knowledge-left');
            
            clickableElements.forEach(element => {
                // 移除可能存在的重复事件监听器
                element.removeEventListener('click', handleClick);
                element.removeEventListener('touchstart', handleTouchStart);
                element.removeEventListener('touchend', handleTouchEnd);
                element.removeEventListener('touchcancel', handleTouchCancel);
                
                // 添加点击事件
                element.addEventListener('click', handleClick);
                
                // 添加触摸事件（使用passive: true以提高性能，但touchend中仍可preventDefault）
                element.addEventListener('touchstart', handleTouchStart, { passive: true });
                element.addEventListener('touchend', handleTouchEnd, { passive: false });
                element.addEventListener('touchcancel', handleTouchCancel, { passive: true });
                
                // 防止长按出现菜单
                element.addEventListener('contextmenu', function(e) {
                    e.preventDefault();
                    return false;
                });
            });
            
            console.log('触摸事件已设置，元素数量:', clickableElements.length);
        }

        // 处理点击事件（主要用于桌面设备）
        function handleClick(e) {
            const element = e.currentTarget;
            const action = element.dataset.action;
            
            console.log('点击事件触发，动作:', action);
            
            // 防止在触摸设备上重复触发（触摸事件已经处理了）
            if (element.touchHandled) {
                element.touchHandled = false;
                return;
            }
            
            // 执行动作
            executeAction(action);
            
            // 添加视觉反馈
            element.classList.add('touch-active');
            setTimeout(() => {
                element.classList.remove('touch-active');
            }, 150);
        }
        
        // 处理触摸开始事件
        function handleTouchStart(e) {
            const element = e.currentTarget;
            element.classList.add('touch-active');
            
            // 记录触摸开始信息
            element.touchStart = {
                x: e.touches[0].clientX,
                y: e.touches[0].clientY,
                time: Date.now()
            };
            
            console.log('触摸开始，元素:', element.dataset.action);
        }
        
        // 处理触摸结束事件
        function handleTouchEnd(e) {
            const element = e.currentTarget;
            const action = element.dataset.action;
            
            // 移除触摸反馈
            element.classList.remove('touch-active');
            
            // 标记触摸已处理，防止click事件重复触发
            element.touchHandled = true;
            
            // 如果没有触摸开始信息，直接执行动作（兼容某些Android设备）
            if (!element.touchStart) {
                if (action) {
                    e.preventDefault();
                    executeAction(action);
                }
                // 延迟清除标记，防止click事件触发
                setTimeout(() => {
                    element.touchHandled = false;
                }, 300);
                return;
            }
            
            // 计算触摸移动距离和时间
            const touch = e.changedTouches[0];
            const deltaX = Math.abs(touch.clientX - element.touchStart.x);
            const deltaY = Math.abs(touch.clientY - element.touchStart.y);
            const deltaTime = Date.now() - element.touchStart.time;
            
            // 只有在明显滑动时才忽略（移动距离大于100px认为是滑动）
            // 否则都视为点击，直接执行动作
            if (deltaX < 100 && deltaY < 100 && deltaTime < 2000) {
                console.log('触摸结束，执行动作:', action, '移动:', deltaX, deltaY, '时间:', deltaTime);
                e.preventDefault();
                executeAction(action);
            } else {
                console.log('触摸移动过大或时间过长，视为滑动，忽略:', deltaX, deltaY, deltaTime);
            }
            
            // 清除触摸开始信息
            element.touchStart = null;
            
            // 延迟清除标记，防止click事件触发
            setTimeout(() => {
                element.touchHandled = false;
            }, 300);
        }
        
        // 处理触摸取消事件
        function handleTouchCancel(e) {
            const element = e.currentTarget;
            element.classList.remove('touch-active');
            element.touchStart = null;
        }
